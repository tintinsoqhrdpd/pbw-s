            setWeeklyStatuses={setWeeklyStatuses}

            setArchives={setArchives}

            <div className="w-6 h-6 rounded-full border border-[#fbbf24] flex items-center justify-center">
              <div className="w-2 h-2 rounded-full bg-[#fbbf24]"></div>
            </div>
            <span>PINOY BIG SISTER: PLATO EDITION</span>

function HousemateLoginForm({ housemates, setCurrentUser, showAlert }) {
  const [nameInput, setNameInput] = useState('');
  const [codeInput, setCodeInput] = useState('');

  const handleLogin = (e) => {
    e.preventDefault();
    const hm = housemates.find(h => h.name.toLowerCase().trim() === nameInput.toLowerCase().trim() && h.code === codeInput.trim());
    if (!hm) {
      showAlert('Invalid Housemate Name or Unique Access Code.');
      return;
    }
    if (hm.status === 'Evicted' || hm.status === 'Suspended') {
      showAlert('Your account has been evicted or suspended by Big Sister. Nomination and voting privileges are revoked.');
      return;
    }
    setCurrentUser({ role: 'housemate', housemate: hm });
    setNameInput('');
    setCodeInput('');
  };

  return (
    <form onSubmit={handleLogin} className="space-y-4">
      <div className="space-y-1">
        <label className="text-xs font-black uppercase text-[#fbbf24]">Housemate Name</label>
        <input 
          type="text" 
          value={nameInput} 
          onChange={e => setNameInput(e.target.value)} 
          placeholder="e.g. Aurelius Vance" 
          className="w-full bg-blue-900 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#fbbf24] transition"
          required
        />
      </div>
      <div className="space-y-1">
        <label className="text-xs font-black uppercase text-[#fbbf24]">Unique Access Code</label>
        <input 
          type="password" 
          value={codeInput} 
          onChange={e => setCodeInput(e.target.value)} 
          placeholder="••••••" 
          className="w-full bg-blue-900 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#fbbf24] transition"

      </button>
    </form>
  );
}

function HousematePortalContent({ housemates, currentUser, nominations, setNominations, activeRound, showAlert }) {
  const hm = currentUser.housemate;
  const weekKey = `week_${activeRound.week}`;
  const myNomination = nominations[weekKey]?.[hm.id];

  const [firstNominee, setFirstNominee] = useState('');
  const [firstReason, setFirstReason] = useState('');
  const [secondNominee, setSecondNominee] = useState('');
  const [secondReason, setSecondReason] = useState('');

  const handleNominationSubmit = (e) => {
    e.preventDefault();
    if (firstNominee === secondNominee) {
      showAlert('You cannot nominate the same housemate twice.');
      return;
    }
    if (firstNominee === hm.id || secondNominee === hm.id) {
      showAlert('You cannot nominate yourself.');
      return;
    }
    if (!firstReason.trim() || !secondReason.trim()) {
      showAlert('Both nomination reasons are required.');
      return;
    }

    const currentSubmissions = nominations[weekKey] || {};
    if (currentSubmissions[hm.id]) {
      showAlert('You have already submitted your nominations for this weekly round.');
      return;
    }

    setNominations(prev => ({
      ...prev,
      [weekKey]: {
        ...currentSubmissions,
        [hm.id]: {
          first: { to: firstNominee, reason: firstReason, points: 3 },
          second: { to: secondNominee, reason: secondReason, points: 1 }
        }
      }
    }));

    showAlert('Nominations successfully submitted and locked in the administrator vault.');
    setFirstNominee('');
    setFirstReason('');
    setSecondNominee('');
    setSecondReason('');
  };

  return (
    <div className="space-y-6">
      <div className="border-b border-blue-900 pb-4">
        <h3 className="text-lg font-black text-[#fbbf24]">Weekly Online Nominations (Week {activeRound.week})</h3>
        <p className="text-xs text-blue-200">Nominate 2 different housemates (3 points for 1st, 1 point for 2nd with required reasons).</p>
      </div>

      {myNomination ? (
        <div className="bg-blue-900 p-6 rounded-2xl border-2 border-blue-800 space-y-4">
          <div className="text-xs font-black text-emerald-400 flex items-center space-x-2">
            <CheckCircle className="w-5 h-5" />
            <span>Nominations Successfully Submitted & Locked</span>
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 text-sm font-medium">
            <div className="bg-blue-950 p-4 rounded-xl border border-blue-800 space-y-1">
              <span className="text-xs text-[#fbbf24] font-black">1st Nominee (3 pts): {myNomination.first.to}</span>
              <p className="text-xs text-blue-100">Reason: {myNomination.first.reason}</p>
            </div>
            <div className="bg-blue-950 p-4 rounded-xl border border-blue-800 space-y-1">
              <span className="text-xs text-[#fbbf24] font-black">2nd Nominee (1 pt): {myNomination.second.to}</span>
              <p className="text-xs text-blue-100">Reason: {myNomination.second.reason}</p>
            </div>
          </div>
        </div>
      ) : (
        <form onSubmit={handleNominationSubmit} className="space-y-6">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            <div className="space-y-3 bg-blue-900 p-5 rounded-2xl border-2 border-blue-800">
              <label className="text-xs font-black uppercase text-[#fbbf24]">First Nominee (3 Points)</label>
              <select 
                value={firstNominee} 
                onChange={e => setFirstNominee(e.target.value)} 
                className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white font-bold"
                required
              >
                <option value="">Select Housemate</option>
                {housemates.filter(h => h.id !== hm.id && h.status === 'Active').map(h => (
                  <option key={h.id} value={h.id}>{h.name} ({h.id})</option>
                ))}
              </select>
              <textarea 
                value={firstReason} 
                onChange={e => setFirstReason(e.target.value)} 
                placeholder="Enter mandatory nomination reason..." 
                className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl p-4 text-sm text-white h-24 focus:outline-none focus:border-[#fbbf24]"
                required
              />
            </div>

            <div className="space-y-3 bg-blue-900 p-5 rounded-2xl border-2 border-blue-800">
              <label className="text-xs font-black uppercase text-[#fbbf24]">Second Nominee (1 Point)</label>
              <select 
                value={secondNominee} 
                onChange={e => setSecondNominee(e.target.value)} 
                className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white font-bold"
                required
              >
                <option value="">Select Housemate</option>
                {housemates.filter(h => h.id !== hm.id && h.status === 'Active').map(h => (
                  <option key={h.id} value={h.id}>{h.name} ({h.id})</option>
                ))}
              </select>
              <textarea 
                value={secondReason} 
                onChange={e => setSecondReason(e.target.value)} 
                placeholder="Enter mandatory nomination reason..." 
                className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl p-4 text-sm text-white h-24 focus:outline-none focus:border-[#fbbf24]"
                required
              />
            </div>
          </div>

          <button type="submit" className="w-full py-4 bg-[#fbbf24] text-blue-950 font-black rounded-xl hover:bg-amber-400 transition duration-300 uppercase tracking-wider text-base shadow-2xl transform active:scale-95">
            LOCK AND SUBMIT NOMINATIONS
          </button>
        </form>
      )}
    </div>
  );
}

function VotingView({ housemates, activeRound, currentWeekVotes, setVotes, resultsConfig, showAlert, goBack }) {
  const [selectedNominee, setSelectedNominee] = useState('');
  const [voteType, setVoteType] = useState('save'); 

  const nomineeList = housemates.filter(h => activeRound?.nominees?.includes(h.id));

  const handleVoteSubmission = (e) => {
    e.preventDefault();
    if (resultsConfig.votingManuallyClosed) {
      showAlert('Voting has been manually closed by the administrator.');
      return;
    }
    if (!selectedNominee) {
      showAlert('Please select a nominee.');
      return;
    }

    setVotes(prev => {
      const weekNum = activeRound.week;
      const weekVotes = prev[weekNum] || { save: {}, evict: {} };
      const category = voteType === 'save' ? 'save' : 'evict';
      const currentCount = weekVotes[category][selectedNominee] || 0;

      return {
        ...prev,
        [weekNum]: {
          ...weekVotes,
          [category]: {
            ...weekVotes[category],
            [selectedNominee]: currentCount + 1
          }
        }
      };
    });

    showAlert(`Successfully cast your Vote to ${voteType === 'save' ? 'SAVE' : 'EVICT'} for nominee ${selectedNominee}! Thank you.`);
    setSelectedNominee('');
  };

  return (
    <div className="max-w-4xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-10 animate-fadeIn">
      <div className="text-center space-y-3 border-b-2 border-blue-900 pb-6">
        <div className="inline-flex items-center space-x-2 text-blue-950 bg-[#fbbf24] px-4 py-1.5 rounded-full text-xs font-black uppercase tracking-widest shadow-lg">
          <Vote className="w-4 h-4" />
          <span>Official PBB Voting Arena • Week {activeRound.week}</span>
        </div>
        <h1 className="text-3xl sm:text-5xl font-black text-white">SAVE OR EVICT</h1>
        <p className="text-sm text-blue-200 max-w-xl mx-auto font-medium">
          Simultaneous voting is active for Week {activeRound.week}. You may vote repeatedly while voting is open.
        </p>
      </div>

      {resultsConfig.votingManuallyClosed ? (
        <div className="bg-red-500/20 border-3 border-red-500 rounded-3xl p-12 text-center text-red-200 font-bold space-y-2">
          <Lock className="w-12 h-12 text-red-400 mx-auto" />
          <h3 className="text-xl font-black text-white">VOTING IS CURRENTLY CLOSED</h3>
          <p className="text-xs">The administrator has closed voting for this round.</p>
        </div>
      ) : nomineeList.length === 0 ? (
        <div className="bg-blue-950 border-3 border-blue-900 rounded-3xl p-12 text-center text-blue-300 font-bold">
          Voting is currently closed or no nominees have been confirmed for this round.
        </div>
      ) : (
        <form onSubmit={handleVoteSubmission} className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-10 space-y-8 shadow-2xl">
          <div className="space-y-3">
            <label className="text-xs font-black uppercase text-[#fbbf24] tracking-wider">1. Select Voting Category</label>
            <div className="grid grid-cols-2 gap-4">
              <button
                type="button"
                onClick={() => setVoteType('save')}
                className={`py-4 px-4 rounded-2xl font-black text-sm uppercase transition duration-300 shadow-lg transform active:scale-95 ${
                  voteType === 'save' 
                    ? 'bg-emerald-500 text-blue-950 border-4 border-white scale-[1.02]' 
                    : 'bg-blue-900 text-white hover:bg-blue-800'
                }`}
              >
                💚 VOTE TO SAVE
              </button>
              <button
                type="button"
                onClick={() => setVoteType('evict')}
                className={`py-4 px-4 rounded-2xl font-black text-sm uppercase transition duration-300 shadow-lg transform active:scale-95 ${
                  voteType === 'evict' 
                    ? 'bg-red-500 text-white border-4 border-white scale-[1.02]' 
                    : 'bg-blue-900 text-white hover:bg-blue-800'
                }`}
              >
                ❤️ VOTE TO EVICT
              </button>
            </div>
          </div>

          <div className="space-y-3">
            <label className="text-xs font-black uppercase text-[#fbbf24] tracking-wider">2. Choose Nominee</label>
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
              {nomineeList.map((hm, idx) => (
                <div 
                  key={hm.id}
                  onClick={() => setSelectedNominee(hm.id)}
                  className={`cursor-pointer border-3 rounded-2xl p-4 text-center transition duration-300 space-y-2 shadow-lg transform active:scale-95 ${
                    selectedNominee === hm.id 
                      ? 'border-[#fbbf24] bg-blue-900 shadow-2xl scale-105' 
                      : 'border-blue-900 bg-blue-950/80 hover:border-blue-700'
                  }`}
                >
                  <div className="w-16 h-16 mx-auto rounded-full overflow-hidden border-2 border-[#fbbf24]">
                    <img src={hm.photo} alt={hm.name} className="w-full h-full object-cover" />
                  </div>
                  <div>
                    <span className="text-xs font-black text-[#fbbf24]">({String.fromCharCode(65 + idx)})</span>
                    <h4 className="text-base font-black text-white">{hm.name}</h4>
                  </div>
                </div>
              ))}
            </div>
          </div>

          <button 
            type="submit"
            className={`w-full py-5 rounded-2xl font-black uppercase tracking-wider text-base transition duration-300 shadow-2xl transform active:scale-95 ${
              voteType === 'save' 
                ? 'bg-emerald-500 hover:bg-emerald-400 text-blue-950' 
                : 'bg-red-600 hover:bg-red-500 text-white'
            }`}
          >
            SUBMIT VOTE TO {voteType.toUpperCase()}
          </button>
        </form>
      )}
    </div>
  );

function ResultsView({ housemates, activeRound, currentWeekVotes, resultsConfig, goBack }) {
  const nomineeIds = activeRound.nominees || [];
  
  const saveVotesMap = currentWeekVotes.save || {};
  const evictVotesMap = currentWeekVotes.evict || {};

  const totalSaveVotes = Object.values(saveVotesMap).reduce((a, b) => a + b, 0) || 1;
  const totalEvictVotes = Object.values(evictVotesMap).reduce((a, b) => a + b, 0) || 1;

  const calculatedResults = nomineeIds.map((id, idx) => {
    const hm = housemates.find(h => h.id === id);
    const saveCount = saveVotesMap[id] || 0;
    const evictCount = evictVotesMap[id] || 0;

    const savePct = (saveCount / totalSaveVotes) * 100;
    const evictPct = (evictCount / totalEvictVotes) * 100;
    const finalScore = savePct - evictPct;

    return {
      id,
      name: hm ? hm.name : 'Unknown',
      photo: hm ? hm.photo : '',
      letter: String.fromCharCode(65 + idx),
      savePct,
      evictPct,
      finalScore
    };
  }).sort((a, b) => b.savePct - a.savePct); // PBB Gen 11 Tally format typically sorted or ordered by Nominee A, B, C...

  return (
    <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-12 animate-fadeIn">
      <div className="text-center space-y-2 border-b-2 border-blue-900 pb-6">
        <div className="inline-flex items-center space-x-2 text-blue-950 bg-[#fbbf24] px-4 py-1.5 rounded-full text-xs font-black uppercase tracking-widest shadow-lg">
          <Trophy className="w-4 h-4" />
          <span>Official Tally & Eviction Hall • Week {activeRound.week}</span>
        </div>
        <h1 className="text-3xl sm:text-5xl font-black text-white">RESULTS & EVICTION</h1>
        <p className="text-sm text-blue-200 font-medium">
          Official scores calculated via formula: Final Score = Save % − Evict %
        </p>
      </div>

      {/* PBB GEN 11 EXACT TALLY GRAPHIC LAYOUT */}
      <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-10 space-y-8 shadow-2xl relative overflow-hidden">
        <div className="flex flex-col sm:flex-row items-center justify-between border-b-2 border-blue-900 pb-6 gap-4">
          <div>
            <div className="text-xs tracking-widest text-[#fbbf24] font-black uppercase">Official Broadcast Graphic</div>
            <h3 className="text-2xl sm:text-3xl font-black text-white flex items-center space-x-3">
              <Radio className="w-7 h-7 text-[#fbbf24] animate-pulse" />
              <span>TALLY OF VOTES (WEEK {activeRound.week})</span>
            </h3>
          </div>
          <span className="text-xs font-mono font-bold bg-blue-900 text-emerald-400 px-4 py-1.5 rounded-full border border-emerald-500/40">
            {resultsConfig.showPartial ? '🟢 Publicly Visible' : '🔒 Hidden by Admin'}
          </span>
        </div>

        {!resultsConfig.showPartial ? (
          <div className="text-center py-16 text-blue-300 font-bold text-sm">
            The partial tally board is currently hidden by administrator controls.
          </div>
        ) : (
          <div className="space-y-6 max-w-3xl mx-auto">
            <div className="flex justify-between items-center text-xs font-black uppercase text-blue-300 px-4">
              <span className="text-emerald-400">Vote to Save</span>
              <span className="text-[#fbbf24]">Nominee</span>
              <span className="text-red-400">Vote to Evict</span>
            </div>

            {calculatedResults.map((res) => {
              // Color mapping for nominee pill based on index or letter
              const badgeColors = { 'A': 'bg-blue-600', 'B': 'bg-amber-600', 'C': 'bg-red-600', 'D': 'bg-emerald-600' };
              const colorClass = badgeColors[res.letter] || 'bg-blue-600';

              return (
                <div key={res.id} className="bg-blue-900/90 border-2 border-blue-800 rounded-2xl p-4 sm:p-5 flex items-center justify-between shadow-xl gap-2 sm:gap-4 relative overflow-hidden">
                  {/* Left: Save Percentage Bar & Value */}
                  <div className="flex-1 flex items-center justify-end space-x-3">
                    <span className="text-base sm:text-xl font-black text-emerald-400 font-mono">{res.savePct.toFixed(2)}%</span>
                    <div className="hidden sm:block w-32 bg-blue-950 h-3 rounded-full overflow-hidden border border-emerald-500/40">
                      <div className="bg-emerald-400 h-full rounded-full transition-all duration-1000" style={{ width: `${Math.min(res.savePct, 100)}%` }}></div>
                    </div>
                  </div>

                  {/* Center: Nominee Badge with Avatar */}
                  <div className={`px-4 sm:px-6 py-2.5 rounded-2xl ${colorClass} border-2 border-[#fbbf24] flex items-center space-x-3 shadow-lg transform hover:scale-105 transition`}>
                    <span className="text-xl sm:text-2xl font-black text-white">{res.letter}</span>
                    <div className="w-10 h-10 rounded-full overflow-hidden border-2 border-white/80 shadow">
                      <img src={res.photo} alt={res.name} className="w-full h-full object-cover" />
                    </div>
                  </div>

                  {/* Right: Evict Percentage Bar & Value */}
                  <div className="flex-1 flex items-center justify-start space-x-3">
                    <div className="hidden sm:block w-32 bg-blue-950 h-3 rounded-full overflow-hidden border border-red-500/40">
                      <div className="bg-red-500 h-full rounded-full transition-all duration-1000" style={{ width: `${Math.min(res.evictPct, 100)}%` }}></div>
                    </div>
                    <span className="text-base sm:text-xl font-black text-red-400 font-mono">{res.evictPct.toFixed(2)}%</span>

        <div className="flex items-center justify-between border-b-2 border-blue-900 pb-4">

          <span className="text-xs text-blue-300 font-mono font-bold bg-blue-900 px-3 py-1 rounded-full">

          </span>
        </div>

        {!resultsConfig.showFinal ? (
          <div className="text-center py-12 text-blue-300 space-y-3 font-bold">
            <Lock className="w-10 h-10 text-[#fbbf24] mx-auto animate-bounce" />
            <p className="text-sm">Final results have not yet been published by the administrator.</p>
          </div>
        ) : (
          <div className="space-y-4">
            {calculatedResults.map((res, index) => {
              const isEvicted = index === calculatedResults.length - 1 && resultsConfig.evictionPublished;
              return (
                <div key={res.id} className={`flex flex-col sm:flex-row items-center justify-between p-5 rounded-2xl border-3 transition duration-300 ${isEvicted ? 'bg-red-500/20 border-red-500' : 'bg-blue-900 border-blue-800'} gap-4 shadow-xl`}>
                  <div className="flex items-center space-x-4">
                    <span className="w-10 h-10 rounded-full bg-[#fbbf24] text-blue-950 font-black text-base flex items-center justify-center shadow">
                      #{index + 1}
                    </span>
                    <img src={res.photo} alt={res.name} className="w-14 h-14 rounded-full object-cover border-2 border-[#fbbf24]" />
                    <div>
                      <h4 className="font-black text-white text-lg">{res.name}</h4>
                      <p className="text-xs font-mono text-blue-200">Save: {res.savePct.toFixed(2)}% | Evict: {res.evictPct.toFixed(2)}%</p>
                    </div>
                  </div>
                  <div className="flex items-center space-x-6">
                    <div className="text-right">
                      <div className="text-xs text-blue-300 font-black uppercase">Final Score</div>
                      <div className="text-xl font-black text-[#fbbf24] font-mono">{res.finalScore.toFixed(2)}</div>
                    </div>
                    {isEvicted && (
                      <span className="bg-red-600 text-white text-xs font-black px-4 py-2 rounded-xl uppercase tracking-wider animate-pulse shadow-lg">
                        EVICTED
                      </span>
                    )}
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </div>
  );
}

function AdminDashboard({ housemates, housematesList, rounds, setRounds, announcements, setAnnouncements, resultsConfig, setResultsConfig, votes, setVotes, currentUser, setCurrentUser, nominations, setNominations, activeRound, weeklyStatuses, setWeeklyStatuses, archives, setArchives, showAlert, goBack }) {
  const [adminPassword, setAdminPassword] = useState('');
  const [activeAdminTab, setActiveAdminTab] = useState('leaderboard'); 

  const [editingHmId, setEditingHmId] = useState(null);
  const [nameInput, setNameInput] = useState('');
  const [bioInput, setBioInput] = useState('');
  const [photoInput, setPhotoInput] = useState('');
  const [codeInput, setCodeInput] = useState('');

  const fileInputRef = useRef(null);

  const handleAdminLogin = (e) => {
    e.preventDefault();
    if (adminPassword === '2009CAPYBARA18') {
      setCurrentUser({ role: 'admin' });
    } else {
      showAlert('Incorrect administrator password.');
    }
  };

  if (!currentUser || currentUser.role !== 'admin') {
    return (
      <div className="max-w-md mx-auto px-4 py-20 animate-fadeIn">
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-8 shadow-2xl space-y-6">
          <div className="text-center space-y-3">
            <div className="w-14 h-14 rounded-full bg-[#fbbf24] flex items-center justify-center text-blue-950 mx-auto shadow-lg">
              <Lock className="w-7 h-7" />
            </div>
            <h2 className="text-2xl font-black text-white">ADMINISTRATOR LOGIN</h2>
            <p className="text-xs text-blue-200 font-medium">Restricted secure management gateway.</p>
          </div>

          <form onSubmit={handleAdminLogin} className="space-y-4">
            <div className="space-y-1">
              <label className="text-xs font-black uppercase text-[#fbbf24]">Admin Password</label>
              <input 
                type="password" 
                value={adminPassword} 
                onChange={e => setAdminPassword(e.target.value)} 
                placeholder="Enter password..." 
                className="w-full bg-blue-900 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white focus:outline-none focus:border-[#fbbf24] transition"
                required
              />
            </div>
            <button type="submit" className="w-full py-4 bg-[#fbbf24] text-blue-950 font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg transform active:scale-95">
              AUTHENTICATE ADMIN
            </button>
          </form>
        </div>
      </div>
    );
  }

  const weekKey = `week_${activeRound.week}`;
  const currentWeekNominations = nominations[weekKey] || {}; 

  const nominationTally = {};
  housematesList.forEach(h => {
    nominationTally[h.id] = { id: h.id, name: h.name, points: 0, receivedFrom: [] };
  });

  Object.entries(currentWeekNominations).forEach(([nominatorId, sub]) => {
    const nominator = housematesList.find(h => h.id === nominatorId);
    const nominatorName = nominator ? nominator.name : nominatorId;

    if (sub.first && nominationTally[sub.first.to]) {
      nominationTally[sub.first.to].points += sub.first.points;
      nominationTally[sub.first.to].receivedFrom.push({
        from: nominatorName,
        points: sub.first.points,
        reason: sub.first.reason
      });
    }
    if (sub.second && nominationTally[sub.second.to]) {
      nominationTally[sub.second.to].points += sub.second.points;
      nominationTally[sub.second.to].receivedFrom.push({
        from: nominatorName,
        points: sub.second.points,
        reason: sub.second.reason
      });
    }
  });

  const sortedLeaderboard = Object.values(nominationTally).sort((a, b) => b.points - a.points);

  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onloadend = () => {
        setPhotoInput(reader.result);
      };
      reader.readAsDataURL(file);
    }
  };

  const handleSaveHousemate = (e) => {
    e.preventDefault();
    if (editingHmId) {
      housemates(prev => prev.map(h => h.id === editingHmId ? { ...h, name: nameInput, bio: bioInput, photo: photoInput || h.photo, code: codeInput } : h));
      setEditingHmId(null);
      showAlert('Housemate profile updated successfully.');
    } else {
      const newId = `HM-00${housematesList.length + 1}`;
      const newHm = {
        id: newId,
        name: nameInput,
        bio: bioInput,
        photo: photoInput || 'https://placehold.co/400x500/0284c7/ffffff?text=Housemate',
        status: 'Active',
        code: codeInput || '12345'
      };
      housemates(prev => [...prev, newHm]);
      showAlert(`Housemate ${newId} added successfully.`);
    }
    setNameInput('');
    setBioInput('');
    setPhotoInput('');
    setCodeInput('');
  };

  const startEdit = (hm) => {
    setEditingHmId(hm.id);
    setNameInput(hm.name);
    setBioInput(hm.bio);
    setPhotoInput(hm.photo);
    setCodeInput(hm.code);
  };

  const handleStatusChange = (id, newStatus) => {
    housemates(prev => prev.map(h => {
      if (h.id === id) {
        return { ...h, status: newStatus };
      }
      return h;
    }));

    if (newStatus === 'Evicted') {
      setWeeklyStatuses(prev => ({
        ...prev,
        [id]: { ...prev[id], [activeRound.week]: 'Evicted' }
      }));
    }

    showAlert(`Housemate status updated to ${newStatus}. Privileges automatically adjusted.`);
  };

  const handleStartNewWeek = () => {
    const nextWeekNum = activeRound.week + 1;
    
    setArchives(prev => [
      { week: activeRound.week, title: `Week ${activeRound.week} Competition & Results`, summary: `Completed week ${activeRound.week} cycle with official nominees and eviction results.` },
      ...prev
    ]);

    setRounds(prev => [...prev, { id: `round-${nextWeekNum}`, week: nextWeekNum, status: 'Voting Open', method: 'Online', confirmed: false, nominees: [] }]);

    setVotes(prev => ({
      ...prev,
      [nextWeekNum]: { save: {}, evict: {} }
    }));

    setResultsConfig({ showPartial: true, showFinal: false, evictionPublished: false, votingManuallyClosed: false });

    showAlert(`Week ${nextWeekNum} has officially started! Nomination ballots and voting tallies have been reset.`);
  };

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-12 space-y-8 animate-fadeIn">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between border-b-2 border-blue-900 pb-6 gap-4">
        <div>
          <h1 className="text-3xl font-black text-white">ADMINISTRATOR CONTROL PANEL</h1>
          <p className="text-sm text-blue-200 font-medium">Manage housemate statuses, nomination leaderboards, weekly cycles, and results.</p>
        </div>
        <div className="flex items-center space-x-3">
          <button 
            onClick={handleStartNewWeek}
            className="px-5 py-3 bg-emerald-500 hover:bg-emerald-400 text-blue-950 font-black text-xs rounded-xl shadow-xl flex items-center space-x-2 transition duration-300 transform active:scale-95"
          >
            <RotateCcw className="w-4 h-4" />
            <span>START NEXT WEEK (RESET CYCLE)</span>
          </button>
        </div>
      </div>

      <div className="flex flex-wrap gap-2">
        {['leaderboard', 'housemates', 'rounds', 'voting', 'results', 'content'].map(tab => (
          <button
            key={tab}
            onClick={() => setActiveAdminTab(tab)}
            className={`px-4 py-2.5 text-xs font-black uppercase rounded-xl transition duration-300 transform active:scale-95 ${activeAdminTab === tab ? 'bg-[#fbbf24] text-blue-950 shadow-lg' : 'bg-blue-900 text-white hover:bg-blue-800'}`}
          >
            {tab}
          </button>
        ))}
      </div>

      {activeAdminTab === 'leaderboard' && (
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <div className="border-b-2 border-blue-900 pb-4 flex items-center justify-between">
            <div>
              <h3 className="text-2xl font-black text-[#fbbf24]">HOUSEMATE NOMINATION LEADERBOARD (ADMIN SECURE)</h3>
              <p className="text-xs text-blue-200 font-medium">Aggregated points and confidential reasons for Week {activeRound.week}.</p>
            </div>
            <span className="bg-red-500 text-white text-xs font-black px-3 py-1 rounded-full uppercase">Confidential</span>
          </div>

          <div className="space-y-6">
            {sortedLeaderboard.map((entry, index) => (
              <div key={entry.id} className="bg-blue-900 border-2 border-blue-800 rounded-2xl p-6 space-y-4 shadow-xl">
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 border-b border-blue-800 pb-3">
                  <div className="flex items-center space-x-3">
                    <span className="w-8 h-8 rounded-full bg-[#fbbf24] text-blue-950 font-black text-sm flex items-center justify-center">
                      #{index + 1}
                    </span>
                    <div>
                      <h4 className="text-lg font-black text-white">{entry.name}</h4>

                    </div>
                  </div>
                  <div className="bg-blue-950 px-4 py-2 rounded-xl border border-blue-800 text-right">
                    <span className="text-xs text-blue-300 font-bold uppercase">Total Points: </span>
                    <span className="text-xl font-black text-[#fbbf24] font-mono">{entry.points}</span>
                  </div>
                </div>

                {entry.receivedFrom.length === 0 ? (
                  <p className="text-xs text-blue-300 italic">No nomination points received this week.</p>

                    <h5 className="text-xs font-black uppercase text-blue-300">Nomination Ballots Received:</h5>
                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                      {entry.receivedFrom.map((ballot, bIdx) => (

                          <div className="flex justify-between text-xs font-bold text-[#fbbf24]">

                    <span className={`text-[10px] uppercase font-black px-2 py-0.5 rounded-full ${hm.status === 'Active' ? 'bg-emerald-500 text-blue-950' : hm.status === 'Suspended' ? 'bg-amber-500 text-blue-950' : 'bg-red-500 text-white'}`}>{hm.status}</span>
                  </div>
                  <img src={hm.photo} alt={hm.name} className="w-20 h-20 mx-auto rounded-full object-cover border-2 border-[#fbbf24]" />
                  <h4 className="font-black text-white text-lg text-center">{hm.name}</h4>
                  <p className="text-xs text-blue-200 line-clamp-2 font-medium">{hm.bio}</p>
                </div>
                <div className="pt-4 border-t border-blue-900 space-y-3">
                  <div className="flex items-center justify-between">
                    <button onClick={() => startEdit(hm)} className="text-xs font-black text-[#fbbf24] hover:underline flex items-center space-x-1">
                      <Edit2 className="w-3.5 h-3.5" />
                      <span>Edit Profile</span>
                    </button>
                  </div>
                  <div className="space-y-1">
                    <label className="text-[10px] uppercase font-black text-blue-300">Set Status (Revokes Privileges if Evicted/Suspended):</label>
                    <select 
                      value={hm.status}
                      onChange={(e) => handleStatusChange(hm.id, e.target.value)}
                      className="w-full bg-blue-900 border border-blue-700 rounded-xl px-3 py-1.5 text-xs text-white font-bold focus:outline-none focus:border-[#fbbf24]"
                    >
                      <option value="Active">Active (Eligible)</option>
                      <option value="Suspended">Suspended (Revoked)</option>
                      <option value="Evicted">Evicted (Revoked)</option>
                    </select>
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {activeAdminTab === 'rounds' && (
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <h3 className="text-xl font-black text-[#fbbf24]">Active Weekly Round & Nominees Configuration</h3>
          <div className="space-y-4">
            {rounds.map((rnd, idx) => (
              <div key={idx} className="bg-blue-900 border-2 border-blue-800 rounded-2xl p-6 space-y-4">
                <div className="flex items-center justify-between">
                  <h4 className="font-black text-white text-lg">Week {rnd.week} ({rnd.status})</h4>
                  <span className="text-xs font-black text-blue-950 bg-[#fbbf24] px-3 py-1 rounded-full">Method: {rnd.method}</span>
                </div>
                <div className="space-y-2">
                  <label className="text-xs font-black uppercase text-blue-300">Official Nominees (Comma separated IDs)</label>
                  <input 
                    type="text" 
                    defaultValue={rnd.nominees.join(', ')} 
                    onBlur={(e) => {
                      const ids = e.target.value.split(',').map(s => s.trim()).filter(Boolean);
                      setRounds(prev => prev.map((r, i) => i === idx ? { ...r, nominees: ids } : r));

                      setWeeklyStatuses(prev => {
                        const updated = { ...prev };
                        housematesList.forEach(hm => {
                          if (!updated[hm.id]) updated[hm.id] = {};
                          if (ids.includes(hm.id)) {
                            updated[hm.id][rnd.week] = 'Nominated';
                          } else if (updated[hm.id][rnd.week] !== 'Evicted') {
                            updated[hm.id][rnd.week] = 'Safe';
                          }
                        });
                        return updated;
                      });

                      showAlert('Nominee list updated and weekly tracker synchronized.');
                    }}
                    className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white font-bold"
                  />
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {activeAdminTab === 'voting' && (
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between border-b-2 border-blue-900 pb-4 gap-4">
            <div>
              <h3 className="text-xl font-black text-[#fbbf24]">Voting Control & Aggregate Totals</h3>
              <p className="text-xs text-blue-200">Manually open or close public and housemate voting anytime.</p>
            </div>
            <button 
              onClick={() => {
                const newState = !resultsConfig.votingManuallyClosed;
                setResultsConfig(prev => ({ ...prev, votingManuallyClosed: newState }));
                showAlert(`Voting has been manually ${newState ? 'CLOSED' : 'OPENED'} by administrator.`);
              }}
              className={`px-5 py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.votingManuallyClosed ? 'bg-emerald-500 text-blue-950' : 'bg-red-600 text-white'}`}
            >
              {resultsConfig.votingManuallyClosed ? '🟢 OPEN VOTING NOW' : '🔒 CLOSE VOTING MANUALLY'}
            </button>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 pt-2">
            <div className="bg-blue-900 border-2 border-blue-800 rounded-2xl p-6 space-y-4">
              <h4 className="font-black text-emerald-400 text-sm uppercase">Vote to Save Totals</h4>
              <div className="space-y-2 font-mono text-sm font-bold">
                {Object.entries(votes[activeRound.week]?.save || {}).map(([id, count]) => (
                  <div key={id} className="flex justify-between border-b border-blue-800 pb-2">
                    <span className="text-white">{id}</span>
                    <span className="text-emerald-400">{count} votes</span>
                  </div>
                ))}
              </div>
            </div>
            <div className="bg-blue-900 border-2 border-blue-800 rounded-2xl p-6 space-y-4">
              <h4 className="font-black text-red-400 text-sm uppercase">Vote to Evict Totals</h4>
              <div className="space-y-2 font-mono text-sm font-bold">
                {Object.entries(votes[activeRound.week]?.evict || {}).map(([id, count]) => (
                  <div key={id} className="flex justify-between border-b border-blue-800 pb-2">
                    <span className="text-white">{id}</span>
                    <span className="text-red-400">{count} votes</span>
                  </div>
                ))}
              </div>
            </div>
          </div>
        </div>
      )}

      {activeAdminTab === 'results' && (
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <h3 className="text-xl font-black text-[#fbbf24]">Results Publication Controls</h3>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-6">
            <div className="bg-blue-900 p-6 rounded-2xl border-2 border-blue-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Partial Tally Graphic</h4>
                <p className="text-xs text-blue-200 mt-1 font-medium">Control visibility of the partial tally board.</p>
              </div>
              <button 
                onClick={() => setResultsConfig(prev => ({ ...prev, showPartial: !prev.showPartial }))}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.showPartial ? 'bg-emerald-500 text-blue-950' : 'bg-blue-950 text-white border border-blue-700'}`}
              >
                {resultsConfig.showPartial ? 'Visible (Click to Hide)' : 'Hidden (Click to Show)'}
              </button>
            </div>

            <div className="bg-blue-900 p-6 rounded-2xl border-2 border-blue-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Final Results Graphic</h4>
                <p className="text-xs text-blue-200 mt-1 font-medium">Publish final scores and rankings to the public.</p>
              </div>
              <button 
                onClick={() => setResultsConfig(prev => ({ ...prev, showFinal: !prev.showFinal }))}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.showFinal ? 'bg-emerald-500 text-blue-950' : 'bg-blue-950 text-white border border-blue-700'}`}
              >
                {resultsConfig.showFinal ? 'Published (Click to Hide)' : 'Unpublished (Click to Publish)'}
              </button>
            </div>

            <div className="bg-blue-900 p-6 rounded-2xl border-2 border-blue-800 space-y-4 flex flex-col justify-between">
              <div>
                <h4 className="font-black text-white text-base">Eviction Announcement</h4>
                <p className="text-xs text-blue-200 mt-1 font-medium">Mark the lowest-ranking nominee as evicted.</p>
              </div>
              <button 
                onClick={() => {
                  const newEvictionState = !resultsConfig.evictionPublished;
                  setResultsConfig(prev => ({ ...prev, evictionPublished: newEvictionState }));

                  if (newEvictionState && activeRound.nominees.length > 0) {
                    const nomineeIds = activeRound.nominees;
                    const saveMap = votes[activeRound.week]?.save || {};
                    const evictMap = votes[activeRound.week]?.evict || {};
                    const totSave = Object.values(saveMap).reduce((a, b) => a + b, 0) || 1;
                    const totEvict = Object.values(evictMap).reduce((a, b) => a + b, 0) || 1;

                    let lowestScore = 999;
                    let evictedId = null;

                    nomineeIds.forEach(id => {
                      const sPct = ((saveMap[id] || 0) / totSave) * 100;
                      const ePct = ((evictMap[id] || 0) / totEvict) * 100;
                      const score = sPct - ePct;
                      if (score < lowestScore) {
                        lowestScore = score;
                        evictedId = id;
                      }
                    });

                    if (evictedId) {
                      housemates(prev => prev.map(h => h.id === evictedId ? { ...h, status: 'Evicted' } : h));
                      setWeeklyStatuses(prev => ({
                        ...prev,
                        [evictedId]: { ...prev[evictedId], [activeRound.week]: 'Evicted' }
                      }));
                    }
                  }
                }}
                className={`w-full py-3 rounded-xl font-black text-xs uppercase shadow transition duration-300 transform active:scale-95 ${resultsConfig.evictionPublished ? 'bg-red-600 text-white' : 'bg-blue-950 text-white border border-blue-700'}`}
              >
                {resultsConfig.evictionPublished ? 'Eviction Published' : 'Publish Eviction'}
              </button>
            </div>
          </div>
        </div>
      )}

      {activeAdminTab === 'content' && (
        <div className="bg-blue-950 border-4 border-[#fbbf24] rounded-3xl p-6 sm:p-8 space-y-6 shadow-2xl animate-fadeIn">
          <h3 className="text-xl font-black text-[#fbbf24]">Content & Announcements Management</h3>
          <form onSubmit={(e) => {
            e.preventDefault();
            const title = e.target.title.value;
            const content = e.target.content.value;
            setAnnouncements(prev => [{ id: Date.now(), title, date: new Date().toISOString().split('T')[0], content }, ...prev]);
            e.target.reset();
            showAlert('Announcement published successfully.');
          }} className="space-y-4 bg-blue-900 p-6 rounded-2xl border-2 border-blue-800">
            <h4 className="font-black text-white text-sm">Post New Announcement</h4>
            <input type="text" name="title" placeholder="Announcement Title" className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl px-4 py-3 text-sm text-white font-bold" required />
            <textarea name="content" placeholder="Announcement content..." className="w-full bg-blue-950 border-2 border-blue-800 rounded-xl p-4 text-sm text-white h-28 font-medium" required />
            <button type="submit" className="px-6 py-3.5 bg-[#fbbf24] text-blue-950 font-black rounded-xl hover:bg-amber-400 transition duration-300 shadow-lg transform active:scale-95">
              PUBLISH ANNOUNCEMENT
            </button>
          </form>
        </div>
      )}
    </div>
  );
}
